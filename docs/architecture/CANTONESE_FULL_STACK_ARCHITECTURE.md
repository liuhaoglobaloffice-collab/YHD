### Python 实现

```python
# src/voice/cantonese_tts.py

import torch
import numpy as np
from typing import Optional
import onnxruntime as ort

class CantoneseTTS:
    """粤语 TTS 语音合成"""
    
    def __init__(
        self,
        model_path: str = "models/vits_cantonese.onnx",
        device: str = "cuda"  # 'cuda' or 'cpu'
    ):
        self.device = device
        
        # 加载 VITS 模型
        providers = ['CUDAExecutionProvider', 'CPUExecutionProvider'] \
            if device == 'cuda' else ['CPUExecutionProvider']
        
        self.session = ort.InferenceSession(
            model_path,
            providers=providers
        )
        
        # 粤语词典（用于文本预处理）
        self.cantonese_dict = self.load_cantonese_dict()
    
    async def synthesize(
        self,
        text: str,
        voice: str = "female",  # 'male' or 'female'
        accent: str = "guangzhou",  # 'guangzhou' or 'hongkong'
        speed: float = 1.0,
        pitch: float = 1.0
    ) -> bytes:
        """
        粤语语音合成
        
        Args:
            text: 粤语文本
            voice: 男声/女声
            accent: 广州话/香港话
            speed: 语速 (0.8-1.5)
            pitch: 音调 (0.8-1.2)
        
        Returns:
            音频数据 (WAV format)
        """
        
        # 1. 文本预处理
        processed_text = self.preprocess_cantonese_text(text)
        
        # 2. 文本转音素
        phonemes = self.text_to_phonemes(
            processed_text,
            accent=accent
        )
        
        # 3. 选择说话人ID
        speaker_id = self._get_speaker_id(voice, accent)
        
        # 4. VITS 推理
        audio = self.session.run(
            None,
            {
                "phonemes": phonemes,
                "speaker_id": speaker_id,
                "speed": speed,
                "pitch": pitch
            }
        )[0]
        
        # 5. 后处理
        audio = self._post_process_audio(audio)
        
        # 6. 转换为 WAV
        wav_bytes = self._to_wav(audio, sample_rate=22050)
        
        return wav_bytes
    
    def preprocess_cantonese_text(self, text: str) -> str:
        """粤语文本预处理"""
        
        # 1. 繁简转换（如果需要）
        text = self._convert_to_traditional(text)
        
        # 2. 数字转汉字
        text = self._number_to_cantonese(text)
        
        # 3. 英文处理
        text = self._process_english_words(text)
        
        # 4. 语气词规范化
        text = text.replace("嘅嘢", "嘅")
        text = text.replace("咁樣", "咁")
        
        return text
    
    def _get_speaker_id(self, voice: str, accent: str) -> int:
        """获取说话人ID"""
        
        speaker_map = {
            ("female", "guangzhou"): 0,
            ("male", "guangzhou"): 1,
            ("female", "hongkong"): 2,
            ("male", "hongkong"): 3
        }
        
        return speaker_map.get((voice, accent), 0)
```
